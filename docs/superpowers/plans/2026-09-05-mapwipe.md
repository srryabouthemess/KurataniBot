# mapwipe Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Permitir que o `/scorewipe` apague todas as plays de um jogador num mapa de uma vez, gerando **uma** entrada no log de auditoria do servidor em vez de uma por score.

**Architecture:** Canal Redis novo `mapwipe`, com `wipe_map_scores` do lado do bancho.py-ex fazendo o lote inteiro num `UPDATE` e um `post_audit_log` só. O `_rebuild_stats` do servidor é generalizado para uma lista de scores, e o `wipe_score` de hoje passa a chamá-lo com `[score]`. Um endpoint de leitura novo na v1 (`get_player_map_scores`) deixa o bot contar as plays antes de confirmar e conferir depois de publicar. No Discord, a mudança é um botão a mais na confirmação do `/scorewipe`, com segunda confirmação própria.

**Tech Stack:** Node.js 22 + discord.js 14 + node:test (bot); Python 3.11 + FastAPI + databases/MySQL + redis (bancho.py-ex); Docker Compose na VPS.

**Spec:** `docs/superpowers/specs/2026-09-05-mapwipe-design.md`

Em `_rebuild_stats`, a assinatura passa a ser:

```python
    user_id: int,
    mode: GameMode,
```

Acrescentar `from collections.abc import Sequence` aos imports se ele ainda não
estiver lá.

Logo depois do `if stats is None: return 0`, substituir o bloco que calcula
`hits` de um score só por:

```python
    plays_removed = len(wiped)
    tscore_removed = sum(score["score"] for score in wiped)

    # A divisão é POR score, e não sobre a soma: a submissão soma
    # `time_elapsed // 1000` a cada play, então só o inverso exato dela devolve
    # o `playtime` que a conta tinha antes. Somar primeiro e dividir depois
    # devolveria até um segundo a mais por play apagada.
    playtime_removed = sum(score["time_elapsed"] // 1000 for score in wiped)

    hits = 0
    for score in wiped:
        hits += score["n300"] + score["n100"] + score["n50"]
        if mode.as_vanilla in (1, 3):
            # taiko conta geki/katu nas notas grandes acertadas com as duas
            # teclas, mania nos 300 arco-íris e nos 200.
            hits += score["ngeki"] + score["nkatu"]
```

- [ ] **Step 2: Trocar as quatro subtrações do UPDATE**

No dicionário de parâmetros do `UPDATE stats`:

```python

            "tscore": max(0, stats["tscore"] - tscore_removed),
            "total_hits": max(0, stats["total_hits"] - hits),
```

O resto do dicionário (`rscore`, `max_combo`, os contadores de grade, `pp`,
`acc`) não muda: essas colunas já são recalculadas do que sobrou no banco.

Logo abaixo do `wipe_score`:

```python
async def wipe_map_scores(
    user_id: int,
    map_md5: str,
    mode: int,
    admin_id: int = 0,
    reason: str = "",
) -> str:
    """Hide every score a player has on one map, in a single audit entry.

    The batch counterpart of `wipe_score`. Wiping a whole map one score at a
    time works, but each call posts its own audit embed, and ten plays turn the
    log into scrolling. Here the whole set moves in one UPDATE, `stats` is
    rewritten once, and the reader gets one entry with the count.
    """
    db = app.state.services.database
    game_mode = GameMode(mode)

    scores = await db.fetch_all(
        "SELECT s.id, s.status, s.score, s.pp, s.grade, s.n300, s.n100, s.n50, "
        "s.ngeki, s.nkatu, s.time_elapsed, m.id AS map_id, m.filename AS map_name "
        "FROM scores s LEFT JOIN maps m ON m.md5 = s.map_md5 "
        "WHERE s.userid = :user_id AND s.map_md5 = :map_md5 AND s.mode = :mode "
        # `>= 0` covers FAILED (0), SUBMITTED (1) and BEST (2), and leaves out
        # what is already parked on WIPED_SCORE_STATUS. The failed ones are in
        # on purpose: they count towards `plays`, and leaving them standing
        # would keep the map adding attempts after a wipe that announced
        # itself as total.
        "AND s.status >= 0",
        {"user_id": user_id, "map_md5": map_md5, "mode": int(game_mode)},
    )

    if not scores:
        # Not a failure, and the same shape of answer `wipe_score` gives for an
        # already wiped score: saying so instead of running on is what keeps a
        # retry from subtracting `plays` a second time.
        return "no scores"

    target = await app.state.sessions.players.from_cache_or_sql(id=user_id)
    if not target:
        return "user not found"

    score_ids = [score["id"] for score in scores]
    placeholders = ", ".join(f":id_{i}" for i in range(len(score_ids)))
    params: dict[str, Any] = {"status": WIPED_SCORE_STATUS}
    params.update({f"id_{i}": score_id for i, score_id in enumerate(score_ids)})

    await db.execute(
        f"UPDATE scores SET status = :status WHERE id IN ({placeholders})",
        params,
    )

    # No runner-up promotion here, unlike `wipe_score`: that exists because the
    # player still has plays left on the map and the BEST slot cannot sit
    # empty. In a batch there are none left, by definition.

    pp = await _rebuild_stats(user_id, game_mode, scores)

    user_info = await db.fetch_one(
        "SELECT country, priv FROM users WHERE id = :id",
        {"id": user_id},
    )

    if user_info is None:
        return "unknown user id"

    if user_info["priv"] & Privileges.UNRESTRICTED:
        await app.state.services.redis.zadd(
            f"bancho:leaderboard:{int(game_mode)}",
            {str(user_id): pp},
        )

        await app.state.services.redis.zadd(
            f"bancho:leaderboard:{int(game_mode)}:{user_info['country']}",
            {str(user_id): pp},
        )

    if target.is_online:
        await target.stats_from_sql_full()

        if not target.restricted:
            app.state.sessions.players.enqueue(app.packets.user_stats(target))

    admin = await app.state.sessions.players.from_cache_or_sql(id=admin_id)
    admin_name = str(admin) if admin else f"Admin({admin_id})"
    mode_names = {
        0: "vn!std", 1: "vn!taiko", 2: "vn!catch", 3: "vn!mania",
        4: "rx!std", 5: "rx!taiko", 6: "rx!catch",
        8: "ap!std"
    }
    mode_name = mode_names.get(int(game_mode), f"mode {game_mode}")

    log(
        f"{admin_name} wiped {len(scores)} scores of {target} on "
        f"{mode_name} map {map_md5} for: {reason}.",
        Ansi.LRED,
    )

    first = scores[0]
    map_name = first["map_name"] or map_md5
    extra = {
        "Mode": mode_name,
        "Map": (
            f"[{map_name}](https://{app.settings.DOMAIN}/b/{first['map_id']})"
            if first["map_id"]
            else map_name
        ),
        "Scores": str(len(scores)),
        "IDs": _format_score_ids(score_ids),
    }

    post_audit_log(
        action="mapwipe",
        admin_id=admin_id,
        admin_name=admin.name if admin else f"#{admin_id}",
        target_id=user_id,
        target_name=target.name,
        reason=reason,
        extra=extra,
    )

    return "success"
```

```python
# Discord refuses an embed field over 1024 characters, and thirty ids go past
# it. The count is the number that matters and it has its own field, so the
# list is allowed to end in a tail.
AUDIT_FIELD_MAX = 1024


def _format_score_ids(score_ids: list[int]) -> str:
    shown: list[str] = []
    length = 0

    for index, score_id in enumerate(score_ids):
        piece = f"`#{score_id}`"
        tail = f" … and {len(score_ids) - index} more"
        if length + len(piece) + 2 + len(tail) > AUDIT_FIELD_MAX:
            return ", ".join(shown) + tail
        shown.append(piece)
        length += len(piece) + 2

    return ", ".join(shown)
```

```python
async def channel_mapwipe_reciever():
    pubsub = app.state.services.redis.pubsub()
    await pubsub.subscribe("mapwipe")

    log("Subscribed to 'mapwipe' channel.", Ansi.LGREEN)

    try:
        async for message in pubsub.listen():
            if message["type"] == "message":
                data = orjson.loads(message["data"])
                # `id` is the PLAYER here, not the score: the batch is defined
                # by who and which map, and the ids are what it finds.
                id = data["id"]
                md5 = data["md5"]
                mode = data["mode"]
                admin_id = data.get("adminId", 0)
                reason = data.get("reason", "")
                log(f"Received message on 'mapwipe'", Ansi.LBLUE)
                log(
                    f"EX | Map wipe | Player: {id}, Map: {md5}, "
                    f"Mode: {mode}, Admin ID: {admin_id}",
                    Ansi.LBLUE,
                )
                response = await wipe_map_scores(id, md5, mode, admin_id, reason)
                log(f"EX | " + response, Ansi.LBLUE)
    except asyncio.CancelledError:
        log("Channel mapwipe receiver task cancelled.", Ansi.LYELLOW)
    finally:
        await pubsub.unsubscribe("mapwipe")
        log("Unsubscribed from 'mapwipe'.", Ansi.LRED)
```

```python
@router.get("/get_player_map_scores")
async def api_get_player_map_scores(
    user_id: int = Query(..., alias="id", ge=3, le=2_147_483_647),
    map_md5: str = Query(..., alias="md5", min_length=32, max_length=32),
    mode_arg: int = Query(..., alias="mode", ge=0, le=8),
) -> Response:
    """Every score one player has on one map, wiped ones included.

    Deliberately without the `status != -1` filter `get_player_scores` has:
    this endpoint exists so KurataniBot can count what a map wipe is about to
    take and then check that nothing above -1 survived it. Hiding the wiped
    rows would make the second question unanswerable.
    """
    rows = await app.state.services.database.fetch_all(
        "SELECT id, status, mode, score, pp, acc, max_combo, mods, grade, "
        "play_time FROM scores "
        "WHERE userid = :user_id AND map_md5 = :map_md5 AND mode = :mode "
        "ORDER BY pp DESC, score DESC",
        {"user_id": user_id, "map_md5": map_md5, "mode": mode_arg},
    )

    return ORJSONResponse(
        {
            "status": "success",
            "scores": [dict(row) for row in rows],
        },
    )
```

- [ ] **Step 7: Escrever `apagarOMapa` acima do `module.exports`**

```js
/**
 * A segunda confirmação, e a publicação do lote.
 *
 * Mora fora do `execute` porque é a terceira tela de um comando que já tinha
 * duas; deixá-la inline faria a função principal passar de duzentas linhas e
 * misturar três fluxos no mesmo escopo.
 *
 * O clique no botão do lote NÃO publica: ele traz para cá, e é só o confirmar
 * daqui que manda. A ação é maior que a de um score e ganha confirmação
 * própria, com as plays na tela.
 */
async function apagarOMapa(interaction, { s, staff, target, modeLabel, modeNum, reason, alvo, doMapa }) {
  const mapLabel  = alvo.mapLabel ?? s.scorewipe_map_unknown;
  const confirmId = `mapwipe_ok_${interaction.id}`;
  const cancelId  = `mapwipe_no_${interaction.id}`;

  const aviso = new EmbedBuilder()
    .setColor(0xff6666)
    .setTitle(s.mapwipe_confirm_title)
    .setDescription(
      s.mapwipe_confirm_body(target.name, target.id, modeLabel, mapLabel, doMapa.length) + '\n\n' +
      // O teto de descrição do embed é 4096; o corte deixa folga para os dois
      // avisos que vêm depois.
      doMapa.map(item => descrever(item, s)).join('\n\n').slice(0, 2500) + '\n\n' +
      s.mapwipe_includes_failed + '\n\n' +
      s.scorewipe_reversible,
    )
    .setFooter({ text: s.nom_actor(staff.osuName) });

  const botoes = new ActionRowBuilder().addComponents(
    new ButtonBuilder().setCustomId(confirmId).setLabel(s.mapwipe_button_confirm(doMapa.length)).setStyle(ButtonStyle.Danger),
    new ButtonBuilder().setCustomId(cancelId).setLabel(s.wipe_button_cancel).setStyle(ButtonStyle.Secondary),
  );

  const prompt = await interaction.editReply({ embeds: [aviso], components: [botoes] });

  let clique;
  try {
    clique = await prompt.awaitMessageComponent({
      filter: i => i.user.id === interaction.user.id,
      time: CONFIRM_MS,
    });
  } catch {
    return interaction.editReply({ content: s.scorewipe_expired, embeds: [], components: [] });
  }

  await clique.deferUpdate().catch(() => {});

  if (clique.customId === cancelId) {
    return interaction.editReply({ content: s.scorewipe_cancelled, embeds: [], components: [] });
  }

  await daycore.wipeMapScores(target.id, alvo.md5, modeNum, {
    osuId:       staff.osuId,
    discordId:   interaction.user.id,
    discordName: interaction.user.username,
  }, reason);

  const confirmado = await daycore.verifyMapScoresWiped(target.id, alvo.md5, modeNum);

  const registrado = registrarAcao('mapwipe', {
    action: 'mapwipe',
    target: target.id,
    detail: `${target.name} | ${modeLabel} | ${doMapa.length} scores em ${mapLabel} | ${reason} | ` +
            `${doMapa.map(item => item.id).join(',')} | ` +
            (confirmado ? 'confirmado' : 'NAO confirmado'),
    actorDiscordId: interaction.user.id,
    actorOsuId: staff.osuId,
    actorOsuName: staff.osuName,
  });

  const resultado = new EmbedBuilder()
    .setColor(confirmado && registrado ? 0x99ff99 : 0xffcc66)
    .setTitle(s.mapwipe_done_title)
    .setDescription(
      s.mapwipe_done_body(target.name, target.id, doMapa.length, mapLabel) + '\n\n' +
      (confirmado ? s.mapwipe_confirmed : s.mapwipe_unconfirmed) +
      (registrado ? '' : '\n\n' + s.admin_log_failed),
    )
    .setFooter({ text: s.nom_actor(staff.osuName) });

  return interaction.editReply({ embeds: [resultado], components: [] });
}
```
